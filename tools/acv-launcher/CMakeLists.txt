cmake_minimum_required(VERSION 3.21)
project(acv-launcher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
# Qt 6.12 LTS is the pinned line (PLAN_GUI.MD stage C); older 6.x fails configuration.
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets)
qt_add_executable(acv-launcher main.cpp MainWindow.cpp MainWindow.h CliClient.cpp CliClient.h)
target_link_libraries(acv-launcher PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
if(BUILD_TESTING)
  enable_testing()
  find_package(Qt6 6.12 REQUIRED COMPONENTS Test)
  qt_add_executable(test-launcher test_launcher.cpp CliClient.cpp CliClient.h MainWindow.cpp MainWindow.h)
  target_link_libraries(test-launcher PRIVATE Qt6::Core Qt6::Widgets Qt6::Test)
  target_compile_definitions(test-launcher PRIVATE NATIVE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../native-rexglue")
  add_test(NAME launcher COMMAND test-launcher)
endif()
