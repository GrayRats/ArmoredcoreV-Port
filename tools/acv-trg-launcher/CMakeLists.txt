cmake_minimum_required(VERSION 3.24)
project(acv_trg_launcher LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ACV_LAUNCHER_ENABLE_VULKAN "Build the SDL2 Vulkan launcher renderer" ON)
# Windows builds are Vulkan-only: they never import OpenGL. Other platforms keep the OpenGL renderer.
if(WIN32)
  set(ACV_LAUNCHER_OPENGL OFF CACHE BOOL "Build the OpenGL renderer (not on Windows)" FORCE)
else()
  option(ACV_LAUNCHER_OPENGL "Build the OpenGL renderer" ON)
endif()
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
FetchContent_Declare(acv_sdl2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.32.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(acv_sdl2)
set(TRG_LAUNCHER_STANDALONE ON CACHE BOOL "" FORCE)
set(TRG_LAUNCHER_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(TRG_LAUNCHER_IMGUI_VERSION v1.92.9b CACHE STRING "" FORCE)
add_subdirectory(../trg-launcher trg)
find_package(Threads REQUIRED)
add_executable(acv-launcher main.cpp)
target_include_directories(acv-launcher PRIVATE ../rexglue-sdk/thirdparty/inja/third_party/include)
target_compile_definitions(acv-launcher PRIVATE SDL_MAIN_HANDLED)
target_compile_options(acv-launcher PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(acv-launcher PRIVATE Threads::Threads)
if(ACV_LAUNCHER_OPENGL)
  find_package(OpenGL REQUIRED)
  target_link_libraries(acv-launcher PRIVATE trg::launcher_standalone OpenGL::GL)
  target_compile_definitions(acv-launcher PRIVATE ACV_LAUNCHER_HAS_OPENGL=1)
else()
  # Vulkan-only build: TRG's launcher library, SDL2 and Dear ImGui's SDL2 backend. trg::launcher_standalone is not
  # linked here because it brings the OpenGL 3 backend along.
  target_sources(acv-launcher PRIVATE ${TRG_LAUNCHER_IMGUI_DIR}/backends/imgui_impl_sdl2.cpp)
  target_include_directories(acv-launcher PRIVATE ${TRG_LAUNCHER_IMGUI_DIR}/backends)
  target_link_libraries(acv-launcher PRIVATE trg::launcher SDL2::SDL2-static)
endif()
set(ACV_LAUNCHER_RENDERER_VULKAN OFF)
if(ACV_LAUNCHER_ENABLE_VULKAN)
  find_package(Vulkan QUIET)
  if(Vulkan_FOUND OR (WIN32 AND NOT MSVC))
    if(NOT EXISTS "${TRG_LAUNCHER_IMGUI_DIR}/backends/imgui_impl_vulkan.cpp")
      message(FATAL_ERROR "Dear ImGui Vulkan backend is missing from ${TRG_LAUNCHER_IMGUI_DIR}")
    endif()
    target_sources(acv-launcher PRIVATE
      vulkan_renderer.cpp
      ${TRG_LAUNCHER_IMGUI_DIR}/backends/imgui_impl_vulkan.cpp)
    target_include_directories(acv-launcher PRIVATE ${TRG_LAUNCHER_IMGUI_DIR}/backends)
    target_compile_definitions(acv-launcher PRIVATE ACV_LAUNCHER_HAS_VULKAN=1)
    if(Vulkan_FOUND)
      target_link_libraries(acv-launcher PRIVATE Vulkan::Vulkan)
    else()
      # MinGW ships no vulkan-1 import library. The loader DLL (vulkan-1.dll) comes with the GPU driver, so it is
      # linked through an import library built here from the exports of the pinned Vulkan headers.
      set(ACV_VULKAN_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../rexglue-sdk/thirdparty/vulkan-headers/include")
      if(NOT EXISTS "${ACV_VULKAN_HEADERS}/vulkan/vulkan_core.h")
        message(FATAL_ERROR "Vulkan headers not found: ${ACV_VULKAN_HEADERS}")
      endif()
      file(STRINGS "${ACV_VULKAN_HEADERS}/vulkan/vulkan_core.h" _acv_vk_decls REGEX "VKAPI_CALL vk[A-Za-z0-9]+\\(")
      set(_acv_vk_names "")
      foreach(_acv_decl IN LISTS _acv_vk_decls)
        string(REGEX REPLACE ".*VKAPI_CALL (vk[A-Za-z0-9]+)\\(.*" "\\1" _acv_name "${_acv_decl}")
        list(APPEND _acv_vk_names "${_acv_name}")
      endforeach()
      list(REMOVE_DUPLICATES _acv_vk_names)
      list(SORT _acv_vk_names)
      set(_acv_def "LIBRARY vulkan-1.dll\nEXPORTS\n")
      foreach(_acv_name IN LISTS _acv_vk_names)
        string(APPEND _acv_def "  ${_acv_name}\n")
      endforeach()
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/vulkan-1.def" "${_acv_def}")
      find_program(ACV_DLLTOOL NAMES ${CMAKE_C_COMPILER_TARGET}-dlltool x86_64-w64-mingw32-dlltool dlltool REQUIRED)
      set(ACV_VULKAN_IMPORT_LIB "${CMAKE_CURRENT_BINARY_DIR}/libvulkan-1.a")
      add_custom_command(OUTPUT "${ACV_VULKAN_IMPORT_LIB}"
        COMMAND "${ACV_DLLTOOL}" -d "${CMAKE_CURRENT_BINARY_DIR}/vulkan-1.def" -l "${ACV_VULKAN_IMPORT_LIB}" -D vulkan-1.dll
        DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/vulkan-1.def"
        VERBATIM)
      add_custom_target(acv_vulkan_import_lib DEPENDS "${ACV_VULKAN_IMPORT_LIB}")
      add_dependencies(acv-launcher acv_vulkan_import_lib)
      target_include_directories(acv-launcher PRIVATE "${ACV_VULKAN_HEADERS}")
      target_link_libraries(acv-launcher PRIVATE "${ACV_VULKAN_IMPORT_LIB}")
    endif()
    set(ACV_LAUNCHER_RENDERER_VULKAN ON)
    message(STATUS "ACV launcher Vulkan renderer enabled")
  else()
    message(STATUS "Vulkan SDK not found; ACV launcher will use its OpenGL renderer")
  endif()
endif()
if(NOT ACV_LAUNCHER_OPENGL AND NOT ACV_LAUNCHER_RENDERER_VULKAN)
  message(FATAL_ERROR "The launcher has no renderer: enable ACV_LAUNCHER_OPENGL or a Vulkan SDK / MinGW build")
endif()

# Reject moved tags or a different local TRG checkout before compiling ACV.
find_package(Git REQUIRED)
function(acv_require_revision directory expected)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Dependency revision differs: ${directory}: ${actual}")
  endif()
endfunction()
