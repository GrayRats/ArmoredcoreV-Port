# Reject moved tags or a different local TRG checkout before compiling ACV.
find_package(Git REQUIRED)
function(acv_require_revision directory expected)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Dependency revision differs: ${directory}: ${actual}")
  endif()
endfunction()
acv_require_revision("${CMAKE_CURRENT_SOURCE_DIR}/../trg-launcher" b384a28c712c9320eea8f9d50639dec49981b6c1)
acv_require_revision("${acv_sdl2_SOURCE_DIR}" 5d249570393f7a37e037abf22cd6012a4cc56a71)
acv_require_revision("${TRG_LAUNCHER_IMGUI_DIR}" f1cc2ae15e53a861a874c3034aae6798fde194ab)

add_executable(acv-iso-test iso_test.cpp)
target_link_libraries(acv-iso-test PRIVATE trg::launcher Threads::Threads)
add_test(NAME iso COMMAND acv-iso-test)
